      function formatInline(str) {
        if (!str) return '';
        let s = escapeHtmlChars(str);

        // Protect escaped asterisks (e.g. f\* -> f*)
        s = s.replace(/\\(\*)/g, '*');

        // Inline code (`...`)
        s = s.replace(/`([^`]+)`/g, '$1');

        // Bold italic (***...***)
        s = s.replace(/\*\*\*([^*]+)\*\*\*/g, '$1');

        // Bold (**...**)
        s = s.replace(/\*\*([^*]+)\*\*/g, '$1');

        // Italic: only match *word* where neither side is another * and not inside math
        s = s.replace(/(?:^|[^*])\*([^*\s][^*]*?[^*\s])\*(?!\*)/g, (m, p1) => {
          return m.replace(`*${p1}*`, `${p1}`);
        });

        // Formulas: f* = ... or f\* = ... or f* = ...
        s = s.replace(/\b(f\*|f*|f\b\s*[\*∗])(\s*[=<>≤≥≈−+\-]\s*[^,\s<;]+)/gi, '$1$2');

        // Arrows
        s = s.replace(/→|→/g, '→');
        s = s.replace(/⇒|⇒/g, '⇒');

        return s;
      }

      function flushTable() {
        if (currentTable && currentTable.length) {
          blocks.push({ type: 'table', lines: currentTable });
          currentTable = null;
        }
      }

      function flushList() {
        if (currentList && currentList.items && currentList.items.length) {
          blocks.push(currentList);
          currentList = null;
        }
      }

      function flushAll() {
        flushQuote();
        flushTable();
        flushList();
      }

      for (let i = 0; i < rawLines.length; i++) {
        const line = rawLines[i];
        const trimmed = line.trim();

        // Code block
        if (trimmed.startsWith('```')) {
          if (currentCode) {
            blocks.push({ type: 'code', lines: currentCode.lines, lang: currentCode.lang });
            currentCode = null;
          } else {
            flushAll();
            currentCode = { lang: trimmed.slice(3).trim(), lines: [] };
          }
          continue;
        }

        if (currentCode) {
          currentCode.lines.push(line);
          continue;
        }

        // Table line
        if (trimmed.startsWith('|') && trimmed.endsWith('|') && trimmed.length > 2) {
          flushQuote();
          flushList();
          if (!currentTable) currentTable = [];
          currentTable.push(trimmed);
          continue;
        } else {
          flushTable();
        }

        // Blockquote
        if (trimmed.startsWith('>') || trimmed.startsWith('>')) {
          flushList();
          if (!currentQuote) currentQuote = [];
          currentQuote.push(trimmed.replace(/^(>|>)\s?/, ''));
          continue;
        } else {
          flushQuote();
        }

        // Horizontal rule
        if (/^(\-{3,}|\_{3,}|\*{3,})$/.test(trimmed)) {
          flushAll();
          blocks.push({ type: 'hr' });
          continue;
        }

        // Major Section Banner (1. 🎯 / 2. 🔍 / 3. 💡 / ### 1. 🎯...)
        const secInfo = isMajorSection(trimmed);
        if (secInfo) {
          flushAll();
          blocks.push({
            type: 'majorSection',
            num: secInfo.num,
            icon: secInfo.icon,
            title: secInfo.title
          });
          continue;
        }

        // Headings: ####, ###, ##, #
        const hMatch = trimmed.match(/^(#{1,6})\s+(.+)$/);
        if (hMatch) {
          flushAll();
          blocks.push({ type: 'heading', level: hMatch[1].length, text: hMatch[2].trim() });
          continue;
        }

        // Unordered list (- item, * item, • item)
        const ulMatch = line.match(/^(\s*)([-*•])\s+(.+)$/);
        if (ulMatch && !/^\s*\*\s*\*/.test(line)) {
          flushQuote();
          if (!currentList || currentList.type !== 'ul') {
            flushList();
            currentList = { type: 'ul', items: [] };
          }
          currentList.items.push(ulMatch[3]);
          continue;
        }

        // Numbered list (1. item, 2. item)
        const olMatch = line.match(/^(\s*)(\d+)[\.\)]\s+(.+)$/);
        if (olMatch && !secInfo) {
          flushQuote();
          if (!currentList || currentList.type !== 'ol') {
            flushList();
            currentList = { type: 'ol', items: [] };
          }
          currentList.items.push(olMatch[3]);
          continue;
        }

        // Empty line
        if (!trimmed) {
          flushAll();
          continue;
        }

        // Headline paragraph check (e.g. ⚖️ PHÁN QUYẾT... or 🎲 BẢN ĐỊNH LƯỢNG...)
        const isLeadHeadline = /^([\p{Extended_Pictographic}\uFE0F]+)\s+([A-ZÀ-Ỹ0-9\s—:–\-\(\)]{4,})/u.test(trimmed);

        // Regular paragraph
        flushAll();
        blocks.push({ type: 'p', text: trimmed, isLead: isLeadHeadline });
      }

      flushAll();
      if (currentCode) {
        blocks.push({ type: 'code', lines: currentCode.lines, lang: currentCode.lang });
      }

      // Render blocks to HTML
      let outHtml = '';
      blocks.forEach(b => {
        switch (b.type) {
          case 'majorSection': {
            const themeClass = b.num === '1' ? 'sec-red' : (b.num === '2' ? 'sec-blue' : 'sec-emerald');
            outHtml += `
              

                

                  PHẦN ${b.num}
                  ${b.icon ? `${b.icon}` : ''}
                

                
${formatInline(b.title)}

              

            `;
            break;
          }
          case 'heading': {
            const hTag = `h${Math.min(b.level, 6)}`;
            const extraClass = b.level === 4 ? 'ai-subheading-h4' : 'ai-heading';
            outHtml += `<${hTag} class="${extraClass}">${formatInline(b.text)}`;
            break;
          }
          case 'table': {
            outHtml += renderTable(b.lines);
            break;
          }
          case 'quote': {
            const quoteContent = b.lines.map(l => `
${formatInline(l)}
`).join('');
            outHtml += `
${quoteContent}
`;
            break;
          }
          case 'ul': {
            outHtml += '
    ';
            b.items.forEach(it => {
              outHtml += `
  • ${formatInline(it)}
    • `;
            });
            outHtml += '
';
            break;
          }
          case 'ol': {
            outHtml += '
    ';
            b.items.forEach(it => {
              outHtml += `
  1. ${formatInline(it)}
    2. `;
            });
            outHtml += '
';
            break;
          }
          case 'code': {
            outHtml += `
${escapeHtmlChars(b.lines.join('\n'))}
`;
            break;
          }
          case 'hr': {
            outHtml += '
';
            break;
          }
          case 'p': {
            const cls = b.isLead ? 'ai-paragraph ai-lead-headline' : 'ai-paragraph';
            outHtml += `
${formatInline(b.text)}
`;
            break;
          }
        }
      });

      return reasoningBlock + outHtml;
    }
